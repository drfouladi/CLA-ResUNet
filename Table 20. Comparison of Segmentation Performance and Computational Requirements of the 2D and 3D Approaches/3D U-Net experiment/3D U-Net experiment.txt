def extract_id(path):

    m = re.search(
        r"(\d+)",
        path.stem
    )

            m.group(1)
        )

    return None

# =============================================================================
# 8. CACHED MMAP HANDLES
# =============================================================================
#
# IMPORTANT V14.1 CHANGE
#
# V14:
#
#     np.load(volume_path, mmap_mode="r")
#     np.load(label_path, mmap_mode="r")
#
# was executed repeatedly for every patch.
#
# V14.1:
#
#     each volume and label is opened once
#
# The mmap itself does NOT load the complete CT into RAM.
#
# =============================================================================

VOLUME_MEMMAPS = {}
LABEL_MEMMAPS = {}


def get_volume(vid):

    vid = int(vid)

    if vid not in VOLUME_MEMMAPS:

        VOLUME_MEMMAPS[vid] = np.load(
            volume_file_map_local[vid],
            mmap_mode="r"
        )

    return VOLUME_MEMMAPS[vid]


def get_label(vid):

    vid = int(vid)

    if vid not in LABEL_MEMMAPS:

        LABEL_MEMMAPS[vid] = np.load(
            label_file_map_local[vid],
            mmap_mode="r"
        )

    return LABEL_MEMMAPS[vid]

# 10. PATCH NORMALIZATION

def normalize_patch(p):

    if isinstance(
        p,
        np.ndarray
    ):

        p = p.tolist()


    if isinstance(
        p,
        dict
    ):

        vid = p.get(
            "volume_id",
            p.get(
                "vid"
            )
        )

        z0 = p.get(
            "z0",
            0
        )

        y0 = p.get(
            "y0",
            0
        )

        x0 = p.get(
            "x0",
            0
        )

        return (
            int(vid),
            int(z0),
            int(y0),
            int(x0)
        )


    if isinstance(
        p,
        (tuple, list)
    ):

        if len(p) != 4:

            raise ValueError(
                f"Invalid patch: {p}"
            )

        return (
            int(p[0]),
            int(p[1]),
            int(p[2]),
            int(p[3])
        )


    raise TypeError(
        f"Unsupported patch type: {type(p)}"
    )

# 11. PATCH VALIDITY
# =============================================================================

def patch_valid(patch):

    vid, z0, y0, x0 = normalize_patch(
        patch
    )

    if vid not in common_ids:

        return False

    H, W, Z = get_volume_shape(
        vid
    )

    if z0 < 0 or y0 < 0 or x0 < 0:

        return False

    if z0 >= Z:

        return False

    if y0 >= H:

        return False

    if x0 >= W:

        return False

    return True


# =============================================================================
# 12. PATCH LOADER
# =============================================================================

def load_patch(patch):

    vid, z0, y0, x0 = normalize_patch(
        patch
    )

    vol = get_volume(
        vid
    )

    lab = get_label(
        vid
    )

    H, W, Z = vol.shape

    z1 = min(
        z0 + DEPTH,
        Z
    )

    y1 = min(
        y0 + PATCH_H,
        H
    )

    x1 = min(
        x0 + PATCH_W,
        W
    )

    image = np.zeros(
        (
            DEPTH,
            PATCH_H,
            PATCH_W
        ),
        dtype=np.float32
    )

    mask = np.zeros(
        (
            DEPTH,
            PATCH_H,
            PATCH_W
        ),
        dtype=np.uint8
    )


    if (
        z1 > z0
        and y1 > y0
        and x1 > x0
    ):

        image_raw = np.asarray(
            vol[
                y0:y1,
                x0:x1,
                z0:z1
            ],
            dtype=np.float32
        )

        mask_raw = np.asarray(
            lab[
                y0:y1,
                x0:x1,
                z0:z1
            ]
        )


        image[
            :z1-z0,
            :y1-y0,
            :x1-x0
        ] = image_raw.transpose(
            2,
            0,
            1
        )


        mask[
            :z1-z0,
            :y1-y0,
            :x1-x0
        ] = mask_raw.transpose(
            2,
            0,
            1
        ).astype(
            np.uint8
        )


    return image, mask

# 13. PATCH GRID
# =============================================================================

def make_patch_grid(ids):

    patches = []

    total_ids = len(ids)

    for index, vid in enumerate(ids):

        H, W, Z = get_volume_shape(
            vid
        )


        z_positions = list(
            range(
                0,
                max(
                    1,
                    Z - DEPTH + 1
                ),
                PATCH_STRIDE_Z
            )
        )


        y_positions = list(
            range(
                0,
                max(
                    1,
                    H - PATCH_H + 1
                ),
                PATCH_STRIDE_Y
            )
        )


        x_positions = list(
            range(
                0,
                max(
                    1,
                    W - PATCH_W + 1
                ),
                PATCH_STRIDE_X
            )
        )


        # Final boundary positions.

        z_last = max(
            0,
            Z - DEPTH
        )

        y_last = max(
            0,
            H - PATCH_H
        )

        x_last = max(
            0,
            W - PATCH_W
        )


        if z_last not in z_positions:

            z_positions.append(
                z_last
            )


        if y_last not in y_positions:

            y_positions.append(
                y_last
            )


        if x_last not in x_positions:

            x_positions.append(
                x_last
            )


        for z0 in z_positions:

            for y0 in y_positions:

                for x0 in x_positions:

                    patches.append(
                        (
                            int(vid),
                            int(z0),
                            int(y0),
                            int(x0)
                        )
                    )


        if (
            index + 1
        ) % 5 == 0:

            print(
                f"Patch generation: "
                f"{index+1}/{total_ids} volumes | "
                f"patches={len(patches):,}"
            )


    return patches

        patch

        mask == 1

    return "background"

def compute_label_distribution(ids):

    counts = {
        0: 0,
        1: 0,
        2: 0
    }

    for vid in sorted(ids):

        lab = get_label(
            vid
        )

        vals, cnts = np.unique(
            lab,
            return_counts=True
        )

        for v, c in zip(

            v = int(v)

            if v in counts:

    return counts

# 23. AUGMENTATION
# =============================================================================

def augment_patch(
    image,
    mask
):

    # Horizontal flip.

    if np.random.random() < 0.50:

        image = image[
            :,
            :,
            ::-1
        ]

        mask = mask[
            :,
            :,
            ::-1
        ]


    # Vertical flip.

    if np.random.random() < 0.50:

        image = image[
            :,
            ::-1,
            :
        ]

        mask = mask[
            :,
            ::-1,
            :
        ]


    # Intensity scaling.

    if np.random.random() < 0.50:

        scale = np.random.uniform(
            0.90,
            1.10
        )

        shift = np.random.uniform(
            -0.05,
            0.05
        )

        image = (
            image * scale
            + shift
        )


    # Small Gaussian noise.

    if np.random.random() < 0.20:

        noise = np.random.normal(
            0.0,
            0.015,
            size=image.shape
        ).astype(
            np.float32
        )

        image = (
            image
            + noise
        )


    return image, mask

# 24. NORMALIZATION
# =============================================================================

def normalize_image(image):

    image = np.nan_to_num(
        image,
        nan=0.0,
        posinf=0.0,
        neginf=0.0
    ).astype(
        np.float32
    )


    finite_values = image[
        np.isfinite(image)
    ]


    if finite_values.size == 0:

        return np.zeros_like(
            image,
            dtype=np.float32
        )


    p1 = np.percentile(
        finite_values,
        1.0
    )

    p99 = np.percentile(
        finite_values,
        99.0
    )


    if p99 > p1:

        image = (
            image - p1
        ) / (
            p99 - p1
        )

    else:

        image = (
            image - p1
        )


    image = np.clip(
        image,
        0.0,
        1.0
    )


    return image.astype(
        np.float32
    )

def make_target(mask):

    liver = (
        mask > 0
    ).astype(
        np.float32
    )

    tumor = (
        mask == 2
    ).astype(
        np.float32
    )


    return np.stack(
        [
            liver,
            tumor
        ],
        axis=-1
    ).astype(
        np.float32
    )

class BalancedTumorSequence(
    keras.utils.Sequence
):

    def __init__(
        self,
        tumor_pool,
        liver_pool,
        background_pool,
        steps_per_epoch,
        batch_size=1
    ):

        super().__init__()

        self.tumor_pool = tumor_pool
        self.liver_pool = liver_pool
        self.background_pool = background_pool

        self.steps_per_epoch = (
            int(
                steps_per_epoch
            )
        )

        self.batch_size = (
            int(
                batch_size
            )
        )

        self.rng = np.random.default_rng(
            SEED + 100
        )


    def __len__(self):

        return self.steps_per_epoch


    def choose_patch(self):

        r = self.rng.random()


        if (
            r < TUMOR_PATCH_PROB
            and len(self.tumor_pool) > 0
        ):

            idx = self.rng.integers(
                0,
                len(self.tumor_pool)
            )

            return self.tumor_pool[
                idx
            ]


        if (
            r <
            TUMOR_PATCH_PROB
            +
            LIVER_PATCH_PROB
            and
            len(self.liver_pool) > 0
        ):

            idx = self.rng.integers(
                0,
                len(self.liver_pool)
            )

            return self.liver_pool[
                idx
            ]


        idx = self.rng.integers(
            0,
            len(self.background_pool)
        )

        return self.background_pool[
            idx
        ]


    def __getitem__(
        self,
        index
    ):

        images = np.zeros(
            (
                self.batch_size,
                DEPTH,
                PATCH_H,
                PATCH_W,
                1
            ),
            dtype=np.float32
        )


        targets = np.zeros(
            (
                self.batch_size,
                DEPTH,
                PATCH_H,
                PATCH_W,
                2
            ),
            dtype=np.float32
        )


        for b in range(
            self.batch_size
        ):

            patch = self.choose_patch()

            image, mask = load_patch(
                patch
            )

            image = normalize_image(
                image
            )

            image, mask = augment_patch(
                image,
                mask
            )

            target = make_target(
                mask
            )

            images[
                b,
                ...,
                0
            ] = image

            targets[
                b
            ] = target


        return (
            images,
            targets
        )

class ValidationSequence(
    keras.utils.Sequence
):

    def __init__(
        self,
        patches,
        batch_size=1
    ):

        super().__init__()

        self.patches = list(
            patches
        )

        self.batch_size = (
            int(
                batch_size
            )
        )


    def __len__(self):

        return int(
            math.ceil(
                len(self.patches)
                /
                self.batch_size
            )
        )


    def __getitem__(
        self,
        index
    ):

        start = (
            index
            *
            self.batch_size
        )

        end = (
            index + 1
        ) * self.batch_size


        batch_patches = self.patches[
            start:end
        ]


        bs = len(
            batch_patches
        )


        images = np.zeros(
            (
                bs,
                DEPTH,
                PATCH_H,
                PATCH_W,
                1
            ),
            dtype=np.float32
        )


        targets = np.zeros(
            (
                bs,
                DEPTH,
                PATCH_H,
                PATCH_W,
                2
            ),
            dtype=np.float32
        )


        for b, patch in enumerate(
            batch_patches
        ):

            image, mask = load_patch(
                patch
            )

            image = normalize_image(
                image
            )

            targets[
                b
            ] = make_target(
                mask
            )

            images[
                b,
                ...,
                0
            ] = image


        return (
            images,
            targets
        )

# 29. GROUP NORMALIZATION
# =============================================================================

class GroupNorm(
    layers.Layer
):

    def __init__(
        self,
        groups=8,
        epsilon=1e-5,
        **kwargs
    ):

        super().__init__(
            **kwargs
        )

        self.groups = groups
        self.epsilon = epsilon


    def build(
        self,
        input_shape
    ):

        channels = int(
            input_shape[-1]
        )


        self.groups_actual = min(
            self.groups,
            channels
        )


        while (
            channels
            %
            self.groups_actual
            != 0
            and
            self.groups_actual > 1
        ):

            self.groups_actual -= 1


        self.gamma = self.add_weight(
            name="gamma",
            shape=(channels,),
            initializer="ones",
            trainable=True
        )


        self.beta = self.add_weight(
            name="beta",
            shape=(channels,),
            initializer="zeros",
            trainable=True
        )


        super().build(
            input_shape
        )


    def call(
        self,
        x
    ):

        shape = tf.shape(
            x
        )


        N = shape[0]
        D = shape[1]
        H = shape[2]
        W = shape[3]
        C = shape[4]


        G = self.groups_actual


        x = tf.reshape(
            x,
            [
                N,
                D,
                H,
                W,
                G,
                C // G
            ]
        )


        mean, var = tf.nn.moments(
            x,
            axes=[
                1,
                2,
                3,
                5
            ],
            keepdims=True
        )


        x = (
            x - mean
        ) / tf.sqrt(
            var
            +
            self.epsilon
        )


        x = tf.reshape(
            x,
            [
                N,
                D,
                H,
                W,
                C
            ]
        )


        return (
            x * self.gamma
            +
            self.beta
        )


    def get_config(self):

        config = super().get_config()

        config.update(
            {
                "groups":
                    self.groups,

                "epsilon":
                    self.epsilon
            }
        )

        return config

# 30. CONV BLOCK
# =============================================================================

def conv_block(
    x,
    filters,
    name
):

    x = layers.Conv3D(
        filters,
        3,
        padding="same",
        use_bias=False,
        name=name + "_conv1"
    )(x)


    x = GroupNorm(
        groups=8,
        name=name + "_gn1"
    )(x)


    x = layers.Activation(
        "relu",
        name=name + "_relu1"
    )(x)


    x = layers.Conv3D(
        filters,
        3,
        padding="same",
        use_bias=False,
        name=name + "_conv2"
    )(x)


    x = GroupNorm(
        groups=8,
        name=name + "_gn2"
    )(x)


    x = layers.Activation(
        "relu",
        name=name + "_relu2"
    )(x)


    return x

# 31. BUILD V14.1 MODEL
# =============================================================================

def build_v14_1_model():

    inputs = keras.Input(
        shape=(
            DEPTH,
            PATCH_H,
            PATCH_W,
            1
        ),
        name="ct_input"
    )


    # -------------------------------------------------------------------------
    # Encoder 1
    # -------------------------------------------------------------------------

    e1 = conv_block(
        inputs,
        16,
        "enc1"
    )


    p1 = layers.MaxPool3D(
        pool_size=(
            1,
            2,
            2
        ),
        name="pool1"
    )(e1)


    # -------------------------------------------------------------------------
    # Encoder 2
    # -------------------------------------------------------------------------

    e2 = conv_block(
        p1,
        32,
        "enc2"
    )


    p2 = layers.MaxPool3D(
        pool_size=(
            1,
            2,
            2
        ),
        name="pool2"
    )(e2)


    # -------------------------------------------------------------------------
    # Encoder 3
    # -------------------------------------------------------------------------

    e3 = conv_block(
        p2,
        64,
        "enc3"
    )


    p3 = layers.MaxPool3D(
        pool_size=(
            1,
            2,
            2
        ),
        name="pool3"
    )(e3)


    # -------------------------------------------------------------------------
    # Bottleneck
    # -------------------------------------------------------------------------

    b = conv_block(
        p3,
        96,
        "bottleneck"
    )


    # -------------------------------------------------------------------------
    # Decoder 3
    # -------------------------------------------------------------------------

    u3 = layers.Conv3DTranspose(
        64,
        kernel_size=(
            1,
            2,
            2
        ),
        strides=(
            1,
            2,
            2
        ),
        padding="same",
        name="up3"
    )(b)


    u3 = layers.Concatenate(
        name="skip3"
    )(
        [
            u3,
            e3
        ]
    )


    d3 = conv_block(
        u3,
        64,
        "dec3"
    )


    # -------------------------------------------------------------------------
    # Decoder 2
    # -------------------------------------------------------------------------

    u2 = layers.Conv3DTranspose(
        32,
        kernel_size=(
            1,
            2,
            2
        ),
        strides=(
            1,
            2,
            2
        ),
        padding="same",
        name="up2"
    )(d3)


    u2 = layers.Concatenate(
        name="skip2"
    )(
        [
            u2,
            e2
        ]
    )


    d2 = conv_block(
        u2,
        32,
        "dec2"
    )


    # -------------------------------------------------------------------------
    # Decoder 1
    # -------------------------------------------------------------------------

    u1 = layers.Conv3DTranspose(
        16,
        kernel_size=(
            1,
            2,
            2
        ),
        strides=(
            1,
            2,
            2
        ),
        padding="same",
        name="up1"
    )(d2)


    u1 = layers.Concatenate(
        name="skip1"
    )(
        [
            u1,
            e1
        ]
    )


    d1 = conv_block(
        u1,
        16,
        "dec1"
    )


    # -------------------------------------------------------------------------
    # Liver branch
    # -------------------------------------------------------------------------

    liver_features = layers.Conv3D(
        16,
        3,
        padding="same",
        use_bias=False,
        name="liver_head_conv"
    )(d1)


    liver_features = GroupNorm(
        groups=8,
        name="liver_head_gn"
    )(liver_features)


    liver_features = layers.Activation(
        "relu",
        name="liver_head_relu"
    )(liver_features)


    liver_prob = layers.Conv3D(
        1,
        1,
        activation="sigmoid",
        name="liver_probability"
    )(liver_features)


    # -------------------------------------------------------------------------
    # SOFT LIVER CONDITIONING
    # -------------------------------------------------------------------------
    #
    # Tumor branch receives continuous liver probability.
    #
    # No hard gating.
    #
    # -------------------------------------------------------------------------

    tumor_context = layers.Concatenate(
        name="soft_liver_tumor_context"
    )(
        [
            d1,
            liver_prob
        ]
    )


    tumor_features = layers.Conv3D(
        24,
        3,
        padding="same",
        use_bias=False,
        name="tumor_head_conv1"
    )(tumor_context)


    tumor_features = GroupNorm(
        groups=8,
        name="tumor_head_gn1"
    )(tumor_features)


    tumor_features = layers.Activation(
        "relu",
        name="tumor_head_relu1"
    )(tumor_features)


    tumor_features = layers.Conv3D(
        16,
        3,
        padding="same",
        use_bias=False,
        name="tumor_head_conv2"
    )(tumor_features)


    tumor_features = GroupNorm(
        groups=8,
        name="tumor_head_gn2"
    )(tumor_features)


    tumor_features = layers.Activation(
        "relu",
        name="tumor_head_relu2"
    )(tumor_features)


    tumor_prob = layers.Conv3D(
        1,
        1,
        activation="sigmoid",
        name="tumor_probability"
    )(tumor_features)


    # -------------------------------------------------------------------------
    # Final output
    # -------------------------------------------------------------------------

    outputs = layers.Concatenate(
        axis=-1,
        name="segmentation_output"
    )(
        [
            liver_prob,
            tumor_prob
        ]
    )


    model = keras.Model(
        inputs=inputs,
        outputs=outputs,
        name="V14_1_TumorFocused_3D_UNet"
    )


    return model


model = build_v14_1_model()

# 32. LOSS FUNCTIONS
# =============================================================================

EPS = 1e-6


def soft_dice(
    y_true,
    y_pred
):

    y_true = tf.cast(
        y_true,
        tf.float32
    )

    y_pred = tf.cast(
        y_pred,
        tf.float32
    )


    axes = [
        1,
        2,
        3
    ]


    intersection = tf.reduce_sum(
        y_true * y_pred,
        axis=axes
    )


    denominator = (
        tf.reduce_sum(
            y_true,
            axis=axes
        )
        +
        tf.reduce_sum(
            y_pred,
            axis=axes
        )
    )


    dice = (
        2.0 * intersection
        +
        EPS
    ) / (
        denominator
        +
        EPS
    )


    return tf.reduce_mean(
        dice
    )


# =============================================================================
# 33. WEIGHTED BCE
# =============================================================================

def weighted_bce(
    y_true,
    y_pred,
    positive_weight=1.0
):

    y_true = tf.cast(
        y_true,
        tf.float32
    )


    y_pred = tf.clip_by_value(
        y_pred,
        EPS,
        1.0 - EPS
    )


    loss = -(
        positive_weight
        *
        y_true
        *
        tf.math.log(
            y_pred
        )
        +
        (
            1.0 - y_true
        )
        *
        tf.math.log(
            1.0 - y_pred
        )
    )


    return tf.reduce_mean(
        loss
    )


# =============================================================================
# 34. FOCUSED NEGATIVE LOSS
# =============================================================================
#
# This specifically increases the importance of difficult negative voxels.
#
# If:
#
#     y_true = 0
#     prediction is high
#
# then the penalty becomes stronger.
#
# This directly targets V14's:
#
#     high recall
#     low precision
#
# problem.
#
# =============================================================================

def negative_focal_loss(
    y_true,
    y_pred,
    gamma=2.0
):

    y_true = tf.cast(
        y_true,
        tf.float32
    )


    y_pred = tf.clip_by_value(
        y_pred,
        EPS,
        1.0 - EPS
    )


    negative_mask = (
        1.0 - y_true
    )


    loss = (
        negative_mask
        *
        tf.pow(
            y_pred,
            gamma
        )
        *
        (
            -tf.math.log(
                1.0 - y_pred
            )
        )
    )


    return tf.reduce_mean(
        loss
    )


# =============================================================================
# 35. FOCUSED TVERSKY
# =============================================================================

def tversky_loss(
    y_true,
    y_pred,
    alpha=TV_ALPHA,
    beta=TV_BETA
):

    y_true = tf.cast(
        y_true,
        tf.float32
    )

    y_pred = tf.cast(
        y_pred,
        tf.float32
    )


    axes = [
        1,
        2,
        3
    ]


    tp = tf.reduce_sum(
        y_true * y_pred,
        axis=axes
    )


    fp = tf.reduce_sum(
        (
            1.0 - y_true
        )
        *
        y_pred,
        axis=axes
    )


    fn = tf.reduce_sum(
        y_true
        *
        (
            1.0 - y_pred
        ),
        axis=axes
    )


    score = (
        tp + EPS
    ) / (
        tp
        +
        alpha * fp
        +
        beta * fn
        +
        EPS
    )


    return tf.reduce_mean(
        1.0 - score
    )


def focal_tversky_loss(
    y_true,
    y_pred
):

    loss = tversky_loss(
        y_true,
        y_pred
    )


    return tf.pow(
        loss + EPS,
        TV_GAMMA
    )


# =============================================================================
# 36. OUTSIDE LIVER PENALTY
# =============================================================================

def outside_liver_penalty(
    true_liver,
    tumor_pred
):

    outside = (
        tumor_pred
        *
        (
            1.0 - true_liver
        )
    )


    return tf.reduce_mean(
        outside
    )


# =============================================================================
# 37. LIVER NON-TUMOR FP PENALTY
# =============================================================================
#
# This is especially important.
#
# True liver but NOT tumor:
#
#     liver = 1
#     tumor = 0
#
# High tumor probability here is a false positive.
#
# =============================================================================

def liver_non_tumor_fp_penalty(
    true_liver,
    true_tumor,
    tumor_pred
):

    liver_non_tumor = (
        true_liver
        *
        (
            1.0 - true_tumor
        )
    )


    penalty = (
        liver_non_tumor
        *
        tf.square(
            tumor_pred
        )
    )


    return tf.reduce_mean(
        penalty
    )


# =============================================================================
# 38. TOTAL V14.1 LOSS
# =============================================================================

def v14_1_loss(
    y_true,
    y_pred
):

    true_liver = (
        y_true[
            ...,
            0:1
        ]
    )


    true_tumor = (
        y_true[
            ...,
            1:2
        ]
    )


    pred_liver = (
        y_pred[
            ...,
            0:1
        ]
    )


    pred_tumor = (
        y_pred[
            ...,
            1:2
        ]
    )


    # -------------------------------------------------------------------------
    # LIVER LOSS
    # -------------------------------------------------------------------------

    liver_bce = weighted_bce(
        true_liver,
        pred_liver,
        positive_weight=1.5
    )


    liver_dice_loss = (
        1.0
        -
        soft_dice(
            true_liver,
            pred_liver
        )
    )


    liver_loss = (
        0.35 * liver_bce
        +
        0.65 * liver_dice_loss
    )


    # -------------------------------------------------------------------------
    # TUMOR LOSS
    # -------------------------------------------------------------------------

    tumor_bce = weighted_bce(
        true_tumor,
        pred_tumor,
        positive_weight=TUMOR_POS_WEIGHT
    )


    tumor_dice_loss = (
        1.0
        -
        soft_dice(
            true_tumor,
            pred_tumor
        )
    )


    tumor_tv_loss = focal_tversky_loss(
        true_tumor,
        pred_tumor
    )


    negative_fp_loss = negative_focal_loss(
        true_tumor,
        pred_tumor,
        gamma=2.0
    )


    tumor_loss = (
        0.20 * tumor_bce
        +
        0.25 * tumor_dice_loss
        +
        0.35 * tumor_tv_loss
        +
        0.20 * negative_fp_loss
    )


    # -------------------------------------------------------------------------
    # FALSE POSITIVE PENALTIES
    # -------------------------------------------------------------------------

    outside_penalty = outside_liver_penalty(
        true_liver,
        pred_tumor
    )


    liver_non_tumor_penalty = (
        liver_non_tumor_fp_penalty(
            true_liver,
            true_tumor,
            pred_tumor
        )
    )


    # -------------------------------------------------------------------------
    # TOTAL
    # -------------------------------------------------------------------------

    total = (
        LIVER_LOSS_WEIGHT
        *
        liver_loss
        +
        TUMOR_LOSS_WEIGHT
        *
        tumor_loss
        +
        OUTSIDE_LIVER_WEIGHT
        *
        outside_penalty
        +
        LIVER_NON_TUMOR_FP_WEIGHT
        *
        liver_non_tumor_penalty
    )


    return total

# 39. METRICS
# =============================================================================

def liver_dice_metric(
    y_true,
    y_pred
):

    return soft_dice(
        y_true[..., 0:1],
        y_pred[..., 0:1]
    )


def tumor_dice_metric(
    y_true,
    y_pred
):

    return soft_dice(
        y_true[..., 1:2],
        y_pred[..., 1:2]
    )


def tumor_precision_metric(
    y_true,
    y_pred
):

    yt = (
        y_true[..., 1]
        >
        0.5
    )


    yp = (
        y_pred[..., 1]
        >
        0.5
    )


    tp = tf.reduce_sum(
        tf.cast(
            yt & yp,
            tf.float32
        )
    )


    fp = tf.reduce_sum(
        tf.cast(
            (
                ~yt
            )
            &
            yp,
            tf.float32
        )
    )


    return (
        tp + EPS
    ) / (
        tp
        +
        fp
        +
        EPS
    )


def tumor_recall_metric(
    y_true,
    y_pred
):

    yt = (
        y_true[..., 1]
        >
        0.5
    )


    yp = (
        y_pred[..., 1]
        >
        0.5
    )


    tp = tf.reduce_sum(
        tf.cast(
            yt & yp,
            tf.float32
        )
    )


    fn = tf.reduce_sum(
        tf.cast(
            yt & (~yp),
            tf.float32
        )
    )


    return (
        tp + EPS
    ) / (
        tp
        +
        fn
        +
        EPS
    )

# 48. FAST GLOBAL PREDICTION FUNCTION
# =============================================================================
#
# IMPORTANT V14.1:
#
# We process validation in batches.
#
# V14:
#
#     model.predict() once per patch
#
# V14.1:
#
#     batch prediction
#
# =============================================================================

def collect_predictions_batched(
    model,
    patches,
    batch_size=1,
    verbose_every=50
):

    results = []

    patches = list(
        patches
    )


    total = len(
        patches
    )


    for start in range(
        0,
        total,
        batch_size
    ):

        batch_patches = patches[
            start:
            start + batch_size
        ]


        bs = len(
            batch_patches
        )


        x = np.zeros(
            (
                bs,
                DEPTH,
                PATCH_H,
                PATCH_W,
                1
            ),
            dtype=np.float32
        )


        masks = []


        for b, patch in enumerate(
            batch_patches
        ):

            image, mask = load_patch(
                patch
            )

            image = normalize_image(
                image
            )

            x[
                b,
                ...,
                0
            ] = image

            masks.append(
                mask
            )


        pred = model(
            x,
            training=False
        ).numpy()


        for b, patch in enumerate(
            batch_patches
        ):

            results.append(
                {
                    "patch":
                        patch,

                    "mask":
                        masks[b],

                    "liver_prob":
                        pred[
                            b,
                            ...,
                            0
                        ].astype(
                            np.float32
                        ),

                    "tumor_prob":
                        pred[
                            b,
                            ...,
                            1
                        ].astype(
                            np.float32
                        )
                }
            )


        if (
            (
                start // batch_size
            )
            %
            verbose_every
            == 0
        ):

            print(
                f"Prediction: "
                f"{min(start + batch_size, total):,}"
                f"/{total:,}"
            )


        del x
        del masks
        del pred


    return results

def calculate_metrics(
    results,
    tumor_threshold=0.50,
    liver_threshold=0.50,
    use_liver_gate=False
):

    tumor_tp = 0
    tumor_fp = 0
    tumor_fn = 0

    liver_tp = 0
    liver_fp = 0
    liver_fn = 0


    for r in results:

        true_liver = (
            r["mask"] > 0
        )


        true_tumor = (
            r["mask"] == 2
        )


        pred_liver = (
            r["liver_prob"]
            >=
            liver_threshold
        )


        pred_tumor = (
            r["tumor_prob"]
            >=
            tumor_threshold
        )


        if use_liver_gate:

            pred_tumor = (
                pred_tumor
                &
                pred_liver
            )


        tumor_tp += int(
            np.sum(
                pred_tumor
                &
                true_tumor
            )
        )


        tumor_fp += int(
            np.sum(
                pred_tumor
                &
                (~true_tumor)
            )
        )


        tumor_fn += int(
            np.sum(
                (~pred_tumor)
                &
                true_tumor
            )
        )


        liver_tp += int(
            np.sum(
                pred_liver
                &
                true_liver
            )
        )


        liver_fp += int(
            np.sum(
                pred_liver
                &
                (~true_liver)
            )
        )


        liver_fn += int(
            np.sum(
                (~pred_liver)
                &
                true_liver
            )
        )


    tumor_dice = (
        2.0 * tumor_tp
        /
        (
            2.0 * tumor_tp
            +
            tumor_fp
            +
            tumor_fn
            +
            1e-8
        )
    )


    tumor_precision = (
        tumor_tp
        /
        (
            tumor_tp
            +
            tumor_fp
            +
            1e-8
        )
    )


    tumor_recall = (
        tumor_tp
        /
        (
            tumor_tp
            +
            tumor_fn
            +
            1e-8
        )
    )


    liver_dice = (
        2.0 * liver_tp
        /
        (
            2.0 * liver_tp
            +
            liver_fp
            +
            liver_fn
            +
            1e-8
        )
    )


    liver_precision = (
        liver_tp
        /
        (
            liver_tp
            +
            liver_fp
            +
            1e-8
        )
    )


    liver_recall = (
        liver_tp
        /
        (
            liver_tp
            +
            liver_fn
            +
            1e-8
        )
    )


    return {

        "tumor_dice":
            float(
                tumor_dice
            ),

        "tumor_precision":
            float(
                tumor_precision
            ),

        "tumor_recall":
            float(
                tumor_recall
            ),

        "tumor_tp":
            int(
                tumor_tp
            ),

        "tumor_fp":
            int(
                tumor_fp
            ),

        "tumor_fn":
            int(
                tumor_fn
            ),

        "liver_dice":
            float(
                liver_dice
            ),

        "liver_precision":
            float(
                liver_precision
            ),

        "liver_recall":
            float(
                liver_recall
            ),

        "liver_tp":
            int(
                liver_tp
            ),

        "liver_fp":
            int(
                liver_fp
            ),

        "liver_fn":
            int(
                liver_fn
            )
    }
