# 3. QUANTITATIVE STABILITY ANALYSIS — LAST 20 EPOCHS
# ================================================================

stability_rows = []
weight_data = {}

for name, cfg in methods.items():

    wdf = pd.read_excel(cfg['weights'])

    required_columns = [
        'epoch', 'background', 'liver', 'tumor'
    ]

    missing_columns = [
        col for col in required_columns
        if col not in wdf.columns
    ]

    if missing_columns:
        raise ValueError(
            f'Missing columns in {cfg["weights"]}: {missing_columns}'
        )

    weight_data[name] = wdf
    last20 = wdf.tail(20)

    row = {'Reward Formulation': name}

    for col in ['background', 'liver', 'tumor']:

        mean_val = last20[col].mean()
        std_val = last20[col].std(ddof=1)

        cv_pct = (
            std_val / mean_val * 100
            if mean_val > 1e-8
            else 0.0
        )

        row[f'{col}_mean'] = mean_val
        row[f'{col}_std'] = std_val
        row[f'{col}_CV(%)'] = cv_pct

    row['Mean_CV(%)'] = np.mean([
        row['background_CV(%)'],
        row['liver_CV(%)'],
        row['tumor_CV(%)']
    ])

    stability_rows.append(row)

stab_df = pd.DataFrame(stability_rows)

reward_traj = {}
aligned = {}

for name, cfg in methods.items():

    hdf = pd.read_csv(cfg['history'])

    required_history_columns = [
        'epoch',
        'dice_liver',
        'dice_tumor',
        'precision_liver'
    ]

    missing_columns = [
        col for col in required_history_columns
        if col not in hdf.columns
    ]

    if missing_columns:
        raise ValueError(
            f'Missing columns in {cfg["history"]}: {missing_columns}'
        )

    # If there are multiple rows for an epoch, average them
    if hdf['epoch'].duplicated().any():
        hdf = hdf.groupby(
            'epoch',
            as_index=False
        ).mean(numeric_only=True)

    hdf['reward'] = cfg['reward_fn'](
        hdf['dice_liver'].values,
        hdf['dice_tumor'].values,
        hdf['precision_liver'].values
    )

    wdf = weight_data[name]

    merged = pd.merge(
        wdf[['epoch', 'background', 'liver', 'tumor']],
        hdf[
            [
                'epoch',
                'reward',
                'dice_liver',
                'dice_tumor',
                'precision_liver'
            ]
        ],
        on='epoch',
        how='inner'
    )

    if merged.empty:
        raise ValueError(
            f'No common epochs found between weight and history files '
            f'for method: {name}'
        )

    aligned[name] = merged
    reward_traj[name] = merged

print('\nReward trajectories reconstructed and aligned successfully.')

def moving_average(series, window=5):
    return series.rolling(
        window=window,
        min_periods=1
    ).mean()


def momentum_update(series, beta=0.9):

    out = [series.iloc[0]]

    for i in range(1, len(series)):
        out.append(
            beta * out[-1] + (1 - beta) * series.iloc[i]
        )

    return pd.Series(
        out,
        index=series.index
    )


comparison_rows = []

for name in methods:

    df = aligned[name]
    tumor = df['tumor']

    cv_cla = (
        tumor.tail(20).std()
        / tumor.tail(20).mean()
        * 100
    )

    ma = moving_average(
        tumor,
        window=5
    )

    mom = momentum_update(
        tumor,
        beta=0.85
    )

    cv_ma = (
        ma.tail(20).std()
        / ma.tail(20).mean()
        * 100
    )

    cv_mom = (
        mom.tail(20).std()
        / mom.tail(20).mean()
        * 100
    )

    comparison_rows.append({
        'Method': name,
        'CLA_CV(%)': round(cv_cla, 2),
        'After_MA5_CV(%)': round(cv_ma, 2),
        'After_Momentum_CV(%)': round(cv_mom, 2)
    })

comp_df = pd.DataFrame(comparison_rows)

def parse_final_metrics(file_path):

    if not os.path.isfile(file_path):
        raise FileNotFoundError(
            f'\nReport file not found:\n{file_path}\n'
            'Please check the method-specific folder and report.txt filename.'
        )

    report_text = Path(file_path).read_text(
        encoding='utf-8-sig',
        errors='replace'
    )

    match = re.search(
        r'Final Metrics\s*\(Mean\s*±\s*SD\)\s*=*\s*\n'
        r'(.*?)(?=\n\s*(?:={3,}|$))',
        report_text,
        flags=re.IGNORECASE | re.DOTALL
    )

    if match:
        metrics_text = match.group(1)

    else:

        match = re.search(
            r'Final Metrics\s*\(Mean\s*±\s*SD\)',
            report_text,
            flags=re.IGNORECASE
        )

        if not match:
            raise ValueError(
                f'Final Metrics section not found in:\n{file_path}'
            )

        metrics_text = report_text[match.end():]

    NUM = r'[-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?'

    results = {}

    for line in metrics_text.splitlines():

        m = re.match(
            rf'\s*([A-Za-z0-9_]+)\s*:\s*'
            rf'({NUM})\s*±\s*({NUM})',
            line
        )

        if m:

            metric_name = m.group(1).strip()

            results[metric_name] = {
                'mean': float(m.group(2)),
                'sd': float(m.group(3))
            }

    required = [
        'Liver_DSC',
        'Tumor_DSC',
        'Tumor_Precision',
        'Tumor_Recall'
    ]

    missing = [
        metric for metric in required
        if metric not in results
    ]

    if missing:
        raise ValueError(
            f'Missing metrics in {file_path}: {missing}\n'
            f'Extracted metrics: {list(results.keys())}'
        )

    return results

stability_values = {}

for method in report_files:

    if method not in weight_data:
        raise ValueError(
            f'Weight data for "{method}" is unavailable.'
        )

    wdf = weight_data[method]

    if 'tumor' not in wdf.columns:
        raise ValueError(
            f'The "tumor" column is missing for "{method}".'
        )

    tumor_weights = pd.to_numeric(
        wdf['tumor'].tail(20),
        errors='coerce'
    ).dropna()

    if len(tumor_weights) < 2:
        raise ValueError(
            f'Insufficient tumor weight values for "{method}".'
        )

    mean_weight = tumor_weights.mean()
    std_weight = tumor_weights.std(ddof=1)

    cv = (
        std_weight / abs(mean_weight) * 100
        if abs(mean_weight) > 1e-8
        else np.nan
    )

    stability_values[method] = cv

table12['Tumor Weight CV (last 20)'] = (
    table12['Reward Formulation']
    .map(stability_values)
    .map(lambda x: f'{x:.2f}%' if pd.notna(x) else 'N/A')
)
