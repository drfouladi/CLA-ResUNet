@tf.keras.utils.register_keras_serializable()
class WindowPartition(layers.Layer):
    def __init__(self, window_size, **kwargs):
        super().__init__(**kwargs)
        self.window_size = window_size

    def build(self, input_shape):
        self.input_shape = input_shape
        super().build(input_shape)

    def call(self, x):
        B, H, W, C = K.shape(x)
        effective_window_size = min(self.window_size, H, W)
        if effective_window_size == 0:
            raise ValueError(f"Window size {self.window_size} برای ابعاد ورودی {H}x{W} معتبر نیست")
        
        num_windows_h = H // effective_window_size
        num_windows_w = W // effective_window_size
        
        if num_windows_h == 0 or num_windows_w == 0:
            windows = K.reshape(x, [-1, H, W, C])
        else:
            x = K.reshape(
                x,
                [B, num_windows_h, effective_window_size, num_windows_w, effective_window_size, C]
            )
            x = K.transpose(x, [0, 1, 3, 2, 4, 5])
            windows = K.reshape(x, [-1, effective_window_size, effective_window_size, C])
        return windows

    def compute_output_shape(self, input_shape):
        B, H, W, C = input_shape
        effective_window_size = min(self.window_size, H, W)
        if effective_window_size == 0:
            return (None, H, W, C)
        num_windows = (H // effective_window_size) * (W // effective_window_size)
        return (None, effective_window_size, effective_window_size, C)

    def get_config(self):
        config = super().get_config()
        config.update({'window_size': self.window_size})
        return config

@tf.keras.utils.register_keras_serializable()
class WindowReverse(layers.Layer):
    def __init__(self, window_size, H, W, **kwargs):
        super().__init__(**kwargs)
        self.window_size = window_size
        self.H = H
        self.W = W

    def build(self, input_shape):
        self.input_shape = input_shape
        super().build(input_shape)

    def call(self, windows):
        effective_window_size = min(self.window_size, self.H, self.W)
        if effective_window_size == 0:
            return K.reshape(windows, [-1, self.H, self.W, K.shape(windows)[-1]])
        
        num_windows_h = self.H // effective_window_size
        num_windows_w = self.W // effective_window_size
        C = K.shape(windows)[-1]
        
        if num_windows_h == 0 or num_windows_w == 0:
            x = K.reshape(windows, [-1, self.H, self.W, C])
        else:
            x = K.reshape(
                windows,
                [-1, num_windows_h, num_windows_w, effective_window_size, effective_window_size, C]
            )
            x = K.transpose(x, [0, 1, 3, 2, 4, 5])
            x = K.reshape(x, [-1, self.H, self.W, C])
        return x

    def compute_output_shape(self, input_shape):
        return (None, self.H, self.W, input_shape[-1])

    def get_config(self):
        config = super().get_config()
        config.update({'window_size': self.window_size, 'H': self.H, 'W': self.W})
        return config

@tf.keras.utils.register_keras_serializable()
class WindowAttention(layers.Layer):
    def __init__(self, dim, window_size, num_heads, qkv_bias=True, **kwargs):
        super().__init__(**kwargs)
        self.dim = dim
        self.window_size = window_size
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = self.head_dim ** -0.5

        self.qkv = layers.Dense(dim * 3, use_bias=qkv_bias)
        self.proj = layers.Dense(dim)

    def build(self, input_shape):
        effective_window_size = min(self.window_size, input_shape[1], input_shape[2])
        self.qkv.build(input_shape)
        self.proj.build((None, effective_window_size, effective_window_size, self.dim))
        self.built = True

    def call(self, x):
        B, H, W, C = K.shape(x)
        qkv = self.qkv(x)
        qkv = K.reshape(qkv, [-1, H * W, 3, self.num_heads, self.head_dim])
        qkv = K.transpose(qkv, [2, 0, 3, 1, 4])
        q, k, v = qkv[0], qkv[1], qkv[2]

        q = q * self.scale
        k_transposed = K.transpose(k, [0, 1, 3, 2])
        attn = K.matmul(q, k_transposed)
        attn = K.softmax(attn, axis=-1)
        x = K.matmul(attn, v)
        x = K.transpose(x, [0, 2, 1, 3])
        x = K.reshape(x, [-1, H, W, C])
        x = self.proj(x)
        return x

    def compute_output_shape(self, input_shape):
        return input_shape

    def get_config(self):
        config = super().get_config()
        config.update({
            'dim': self.dim,
            'window_size': self.window_size,
            'num_heads': self.num_heads,
            'qkv_bias': self.qkv.use_bias
        })
        return config

@tf.keras.utils.register_keras_serializable()
class SwinTransformerBlock(layers.Layer):
    def __init__(self, dim, input_resolution, num_heads, window_size=8, shift_size=0, **kwargs):
        super().__init__(**kwargs)
        self.dim = dim
        self.input_resolution = input_resolution
        self.num_heads = num_heads
        self.window_size = window_size
        self.shift_size = shift_size

        self.norm1 = layers.LayerNormalization(epsilon=1e-5)
        self.attn = WindowAttention(
            dim, window_size=window_size, num_heads=num_heads
        )
        self.norm2 = layers.LayerNormalization(epsilon=1e-5)
        self.mlp = tf.keras.Sequential([
            layers.Dense(dim * 4),
            layers.Activation('gelu'),
            layers.Dense(dim)
        ])
        self.window_partition = WindowPartition(window_size=window_size)
        self.window_reverse = WindowReverse(window_size=window_size, H=input_resolution[0], W=input_resolution[1])

    def build(self, input_shape):
        self.norm1.build(input_shape)
        effective_window_size = min(self.window_size, input_shape[1], input_shape[2])
        self.attn.build((None, effective_window_size, effective_window_size, self.dim))
        self.norm2.build(input_shape)
        self.mlp.build(input_shape)
        self.window_partition.build(input_shape)
        self.window_reverse.build((None, effective_window_size, effective_window_size, self.dim))
        self.built = True

    def call(self, x):
        H, W = self.input_resolution
        B, H_, W_, C = K.shape(x)
        H_, W_, C = K.cast(H_, tf.int32), K.cast(W_, tf.int32), K.cast(C, tf.int32)
        tf.debugging.assert_equal(H_, H, message="ارتفاع ورودی نادرست است")
        tf.debugging.assert_equal(W_, W, message="عرض ورودی نادرست است")

        shortcut = x
        x = self.norm1(x)

        effective_window_size = min(self.window_size, H, W)
        if effective_window_size == H and effective_window_size == W:
            x = self.attn(x)
        else:
            if self.shift_size > 0:
                shifted_x = K.roll(x, shift=[-self.shift_size, -self.shift_size], axis=[1, 2])
            else:
                shifted_x = x

            x_windows = self.window_partition(shifted_x)
            attn_windows = self.attn(x_windows)
            shifted_x = self.window_reverse(attn_windows)

            if self.shift_size > 0:
                x = K.roll(shifted_x, shift=[self.shift_size, self.shift_size], axis=[1, 2])
            else:
                x = shifted_x

        x = shortcut + x
        x = x + self.mlp(self.norm2(x))
        return x

    def compute_output_shape(self, input_shape):
        return input_shape

    def get_config(self):
        config = super().get_config()
        config.update({
            'dim': self.dim,
            'input_resolution': self.input_resolution,
            'num_heads': self.num_heads,
            'window_size': self.window_size,
            'shift_size': self.shift_size
        })
        return config

@tf.keras.utils.register_keras_serializable()
class MLP(tf.keras.layers.Layer):
    def __init__(self, dim, mlp_ratio=4.0, **kwargs):
        super(MLP, self).__init__(**kwargs)
        self.dim = dim
        self.mlp_ratio = mlp_ratio

    def build(self, input_shape):
        self.dense1 = tf.keras.layers.Dense(int(self.dim * self.mlp_ratio), activation='gelu', name='mlp_dense1')
        self.dense2 = tf.keras.layers.Dense(self.dim, name='mlp_dense2')
        super(MLP, self).build(input_shape)

    def call(self, x):
        x = self.dense1(x)
        x = self.dense2(x)
        return x

    def get_config(self):
        config = super().get_config()
        config.update({
            'dim': self.dim,
            'mlp_ratio': self.mlp_ratio
        })
        return config

@tf.keras.utils.register_keras_serializable()
class PatchMerging(tf.keras.layers.Layer):
    def __init__(self, dim, **kwargs):
        super(PatchMerging, self).__init__(**kwargs)
        self.dim = dim

    def build(self, input_shape):
        self.dense = tf.keras.layers.Dense(self.dim * 2, name='dense')
        super(PatchMerging, self).build(input_shape)

    def call(self, x):
        B, H, W, C = tf.shape(x)[0], x.shape[1], x.shape[2], x.shape[3]
        x = tf.reshape(x, [B, H // 2, 2, W // 2, 2, C])
        x = tf.transpose(x, [0, 1, 3, 2, 4, 5])
        x = tf.reshape(x, [B, H // 2, W // 2, 4 * C])
        x = self.dense(x)
        return x

    def get_config(self):
        config = super().get_config()
        config.update({'dim': self.dim})
        return config

# تابع بارگذاری و پیش‌پردازش داده
def load_and_preprocess(vol_path, seg_path):
    try:
        vol_data = np.load(vol_path)
        seg_data = np.load(seg_path)
        if vol_data.size == 0 or seg_data.size == 0:
            raise ValueError("فایل خالی است")
        vol_resized = tf.image.resize(vol_data[..., np.newaxis], (128, 128)).numpy()
        seg_resized = tf.image.resize(seg_data[..., np.newaxis], (128, 128), method='nearest').numpy()
        vol_normalized = (vol_resized - vol_resized.min()) / (vol_resized.max() - vol_resized.min() + 1e-6)
        seg_resized = np.round(seg_resized).astype(np.uint8)
        seg_resized = np.clip(seg_resized[..., 0], 0, 2)
        if seg_resized.shape != (128, 128) or vol_normalized.shape != (128, 128, 1):
            raise ValueError(f"ابعاد نادرست: تصویر = {vol_normalized.shape}, ماسک = {seg_resized.shape}")
        return vol_normalized, seg_resized
    except Exception as e:
        print(f"خطا در بارگذاری {vol_path}, {seg_path}: {e}")
        return np.zeros((128, 128, 1)), np.zeros((128, 128))

# تابع پس‌پردازش اصلاح‌شده
def post_process_predictions(pred_probs, tumor_threshold=0.2, liver_min_size=30, tumor_min_size=3):
    if pred_probs.shape[-1] != 3 or len(pred_probs.shape) != 4:
        print(f"خطا: ابعاد نادرست pred_probs: {pred_probs.shape}")
        return np.zeros((pred_probs.shape[0], 128, 128), dtype=np.uint8)
    
    pred_labels = np.argmax(pred_probs, axis=-1).astype(np.uint8)
    tumor_probs = pred_probs[..., 2]
    pred_labels[(pred_labels == 2) & (tumor_probs < tumor_threshold)] = 0
    
    for i in range(pred_labels.shape[0]):
        liver_mask = (pred_labels[i] == 1).astype(np.uint8)
        labeled_liver, num_features = measure.label(liver_mask, return_num=True)
        for label in range(1, num_features + 1):
            region_size = np.sum(labeled_liver == label)
            if region_size < liver_min_size:
                pred_labels[i][labeled_liver == label] = 0
        tumor_mask = (pred_labels[i] == 2).astype(np.uint8)
        labeled_tumor, num_features = measure.label(tumor_mask, return_num=True)
        for label in range(1, num_features + 1):
            region_size = np.sum(labeled_tumor == label)
            if region_size < tumor_min_size:
                pred_labels[i][labeled_tumor == label] = 0
    return pred_labels

# تعریف معیارها
def dice_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(y_pred == class_id, tf.float32)
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    denominator = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f)
    return tf.where(denominator > 0, (2. * intersection + 1e-5) / (denominator + 1e-5), 0.0)

def iou_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(y_pred == class_id, tf.float32)
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f) - intersection
    return tf.where(union > 0, (intersection + 1e-5) / (union + 1e-5), 0.0)

def precision_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(y_pred == class_id, tf.float32)
    true_positives = tf.reduce_sum(y_true_f * y_pred_f)
    predicted_positives = tf.reduce_sum(y_pred_f)
    return tf.where(predicted_positives > 0, true_positives / (predicted_positives + 1e-5), 0.0)

def recall_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(y_pred == class_id, tf.float32)
    true_positives = tf.reduce_sum(y_true_f * y_pred_f)
    actual_positives = tf.reduce_sum(y_true_f)
    return tf.where(actual_positives > 0, true_positives / (actual_positives + 1e-5), 0.0)

def dice_liver(y_true, y_pred):
    return dice_per_class(y_true, y_pred, 1)

def dice_tumor(y_true, y_pred):
    return dice_per_class(y_true, y_pred, 2)

def iou_liver(y_true, y_pred):
    return iou_per_class(y_true, y_pred, 1)

def iou_tumor(y_true, y_pred):
    return iou_per_class(y_true, y_pred, 2)

def precision_liver(y_true, y_pred):
    return precision_per_class(y_true, y_pred, 1)

def precision_tumor(y_true, y_pred):
    return precision_per_class(y_true, y_pred, 2)

def recall_liver(y_true, y_pred):
    return recall_per_class(y_true, y_pred, 1)

def recall_tumor(y_true, y_pred):
    return recall_per_class(y_true, y_pred, 2)

def dice_avg(y_true, y_pred):
    dice_l = dice_liver(y_true, y_pred)
    dice_t = dice_tumor(y_true, y_pred)
    return (dice_l + dice_t) / 2

def focal_loss(gamma=2., alpha=0.25):
    def focal_loss_fixed(y_true, y_pred):
        y_true = tf.one_hot(tf.cast(y_true, tf.int32), depth=3)
        y_pred = tf.clip_by_value(y_pred, 1e-7, 1. - 1e-7)
        ce = -y_true * tf.math.log(y_pred)
        weight = y_true * tf.pow(1 - y_pred, gamma)
        fl = alpha * weight * ce
        return tf.reduce_mean(fl)
    return focal_loss_fixed

def tversky_loss(y_true, y_pred, alpha=0.6, beta=0.4):
    y_true_f = tf.one_hot(tf.cast(y_true, tf.int32), depth=3)
    y_pred_f = tf.nn.softmax(y_pred, axis=-1)
    numerator = tf.reduce_sum(y_true_f * y_pred_f, axis=[1, 2])
    denominator = numerator + alpha * tf.reduce_sum(y_true_f * (1 - y_pred_f), axis=[1, 2]) + \
                  beta * tf.reduce_sum((1 - y_true_f) * y_pred_f, axis=[1, 2])
    tversky = numerator / (denominator + 1e-5)
    return tf.where(denominator > 0, 1 - tf.reduce_mean(tversky), 0.0)

# تعریف HLA
class HierarchicalLearningAutomata:
    def __init__(self, num_classes=3, initial_weights=None, alpha_high=0.2, beta_high=0.1, alpha_low=0.15, beta_low=0.08, liver_priority=1.5, tumor_priority=4.0):
        self.num_classes = num_classes
        self.alpha_high = alpha_high
        self.beta_high = beta_high
        self.alpha_low = alpha_low
        self.beta_low = beta_low
        self.liver_priority = liver_priority
        self.tumor_priority = tumor_priority
        self.weights = np.array(initial_weights) if initial_weights is not None else np.ones(num_classes) * 7.0
        self.probabilities = self.weights / np.sum(self.weights)
        self.tumor_boost = 3.0
        self.history = []
        self.best_weights = self.weights.copy()
        self.best_metric = 0.0
        self.moving_avg_dice_liver = []
        self.moving_avg_dice_tumor = []
        self.window_size = 5
        self.high_level_probs = np.ones(num_classes) / num_classes
        self.low_level_probs = [np.ones(3) / 3 for _ in range(num_classes)]
        self.low_level_actions = {0: 0.5, 1: -0.5, 2: 0.0}

    def select_action(self):
        if len(self.moving_avg_dice_tumor) >= self.window_size and np.mean(self.moving_avg_dice_tumor[-self.window_size:]) < 0.80:
            selected_class = 2
        elif len(self.moving_avg_dice_liver) >= self.window_size and np.mean(self.moving_avg_dice_liver[-self.window_size:]) < 0.90:
            selected_class = 1
        else:
            adjusted_probs = self.high_level_probs.copy()
            adjusted_probs[1] *= self.liver_priority
            adjusted_probs[2] *= self.tumor_priority * self.tumor_boost
            adjusted_probs /= np.sum(adjusted_probs)
            selected_class = np.random.choice(self.num_classes, p=adjusted_probs)
        low_level_action = np.random.choice(3, p=self.low_level_probs[selected_class])
        return selected_class, low_level_action

    def update_weights(self, selected_class, low_level_action, reward, metric, liver_metric, tumor_metric, precision_liver, epoch):
        self.moving_avg_dice_liver.append(liver_metric)
        self.moving_avg_dice_tumor.append(tumor_metric)
        
        if len(self.moving_avg_dice_tumor) >= self.window_size:
            avg_dice_tumor = np.mean(self.moving_avg_dice_tumor[-self.window_size:])
            if avg_dice_tumor < 0.80:
                self.tumor_boost = min(self.tumor_boost + 0.1, 3.0)
            else:
                self.tumor_boost = max(self.tumor_boost - 0.05, 1.0)

        combined_metric = 0.4 * liver_metric + 0.4 * tumor_metric + 0.2 * precision_liver
        if combined_metric > self.best_metric:
            self.best_metric = combined_metric
            self.best_weights = self.weights.copy()

        alpha_high = self.alpha_high * np.exp(-epoch / 50) + 0.01
        beta_high = self.beta_high * np.exp(-epoch / 50) + 0.005
        alpha_low = self.alpha_low * np.exp(-epoch / 50) + 0.008
        beta_low = self.beta_low * np.exp(-epoch / 50) + 0.004

        if reward > 0:
            self.high_level_probs[selected_class] += alpha_high * (1 - self.high_level_probs[selected_class])
            for i in range(self.num_classes):
                if i != selected_class:
                    self.high_level_probs[i] *= (1 - alpha_high)
        else:
            self.high_level_probs[selected_class] = max(0.01, self.high_level_probs[selected_class] - beta_high * self.high_level_probs[selected_class])
            for i in range(self.num_classes):
                if i != selected_class:
                    self.high_level_probs[i] += beta_high * self.high_level_probs[selected_class] / (self.num_classes - 1)
        self.high_level_probs /= np.sum(self.high_level_probs)

        if reward > 0:
            self.low_level_probs[selected_class][low_level_action] += alpha_low * (1 - self.low_level_probs[selected_class][low_level_action])
            for i in range(3):
                if i != low_level_action:
                    self.low_level_probs[selected_class][i] *= (1 - alpha_low)
        else:
            self.low_level_probs[selected_class][low_level_action] = max(0.01, self.low_level_probs[selected_class][low_level_action] - beta_low * self.low_level_probs[selected_class][low_level_action])
            for i in range(3):
                if i != low_level_action:
                    self.low_level_probs[selected_class][i] += beta_low * self.low_level_probs[selected_class][low_level_action] / 2
        self.low_level_probs[selected_class] /= np.sum(self.low_level_probs[selected_class])

        self.weights[selected_class] += self.low_level_actions[low_level_action]
        if selected_class == 2 and tumor_metric < 0.80:
            self.weights[2] += 0.5 * self.tumor_boost
        if selected_class == 1 and liver_metric < 0.90:
            self.weights[1] += 0.5
        self.weights = np.clip(self.weights, 1.0, 10.0)
        self.probabilities = self.weights / np.sum(self.weights)

        self.history.append({
            'weights': self.weights.tolist(),
            'probabilities': self.probabilities.tolist(),
            'metric': float(metric),
            'liver_metric': float(liver_metric),
            'tumor_metric': float(tumor_metric),
            'precision_liver': float(precision_liver),
            'tumor_boost': float(self.tumor_boost),
            'high_level_action': int(selected_class),
            'low_level_action': int(low_level_action),
            'high_level_probs': self.high_level_probs.tolist(),
            'low_level_probs': [probs.tolist() for probs in self.low_level_probs]
        })

    def get_weights(self):
        return self.weights

    def get_best_weights(self):
        return self.best_weights

    def save_history(self, filepath):
        with open(filepath, 'w') as f:
            json.dump(self.history, f, indent=4)

class BayesianLearningAutomata:
    def __init__(self, num_classes=3, initial_weights=None, alpha=0.18, beta=0.09, liver_priority=1.5, tumor_priority=4.0):
        self.num_classes = num_classes
        self.alpha = alpha
        self.beta = beta
        self.liver_priority = liver_priority
        self.tumor_priority = tumor_priority
        self.dirichlet_params = np.array(initial_weights) if initial_weights is not None else np.ones(num_classes) * 7.0
        self.weights = self.dirichlet_params / np.sum(self.dirichlet_params)
        self.probabilities = self.weights.copy()
        self.history = []
        self.best_weights = self.weights.copy()
        self.best_metric = 0.0
        self.moving_avg_dice_liver = []
        self.moving_avg_dice_tumor = []
        self.window_size = 5
        self.tumor_boost = 1.0
        self.success_counts = np.zeros(num_classes)
        self.failure_counts = np.zeros(num_classes)

    def select_action(self):
        if len(self.moving_avg_dice_tumor) >= self.window_size and np.mean(self.moving_avg_dice_tumor[-self.window_size:]) < 0.80:
            return 2
        elif len(self.moving_avg_dice_liver) >= self.window_size and np.mean(self.moving_avg_dice_liver[-self.window_size:]) < 0.90:
            return 1
        else:
            sampled_probs = np.random.dirichlet(self.dirichlet_params)
            adjusted_probs = sampled_probs.copy()
            adjusted_probs[1] *= self.liver_priority
            adjusted_probs[2] *= self.tumor_priority * self.tumor_boost
            adjusted_probs /= np.sum(adjusted_probs)
            return np.random.choice(self.num_classes, p=adjusted_probs)

    def update_weights(self, selected_class, reward, metric, liver_metric, tumor_metric, precision_liver, epoch):
        self.moving_avg_dice_liver.append(liver_metric)
        self.moving_avg_dice_tumor.append(tumor_metric)

        if len(self.moving_avg_dice_tumor) >= self.window_size:
            avg_dice_tumor = np.mean(self.moving_avg_dice_tumor[-self.window_size:])
            if avg_dice_tumor < 0.80:
                self.tumor_boost = min(self.tumor_boost + 0.1, 3.0)
            else:
                self.tumor_boost = max(self.tumor_boost - 0.05, 1.0)

        combined_metric = 0.3 * liver_metric + 0.5 * tumor_metric + 0.2 * precision_liver

        if combined_metric > self.best_metric:
            self.best_metric = combined_metric
            self.best_weights = self.weights.copy()

        alpha = self.alpha * np.exp(-epoch / 50) + 0.01
        beta = self.beta * np.exp(-epoch / 50) + 0.005

        if reward > 0:
            self.success_counts[selected_class] += alpha * (10 - self.dirichlet_params[selected_class])
            if selected_class == 2 and tumor_metric < 0.80:
                self.success_counts[2] += alpha * self.tumor_boost * 1.5
            if selected_class == 1 and liver_metric > 0.90:
                self.failure_counts[0] += alpha * 0.3
        else:
            self.failure_counts[selected_class] += beta * self.dirichlet_params[selected_class]
            if selected_class != 2 and tumor_metric < 0.80:
                self.success_counts[2] += alpha * self.tumor_boost * 0.5
            if selected_class != 1 and liver_metric < 0.90:
                self.success_counts[1] += alpha * 0.3

        self.dirichlet_params = np.clip(self.dirichlet_params + self.success_counts - self.failure_counts, 1.0, 10.0)
        self.weights = self.dirichlet_params / np.sum(self.dirichlet_params)
        self.probabilities = self.weights.copy()

        self.history.append({
            'weights': self.weights.tolist(),
            'dirichlet_params': self.dirichlet_params.tolist(),
            'success_counts': self.success_counts.tolist(),
            'failure_counts': self.failure_counts.tolist(),
            'metric': float(metric),
            'liver_metric': float(liver_metric),
            'tumor_metric': float(tumor_metric),
            'precision_liver': float(precision_liver),
            'tumor_boost': float(self.tumor_boost)
        })

    def get_weights(self):
        return self.weights

    def get_best_weights(self):
        return self.best_weights

    def save_history(self, filepath):
        with open(filepath, 'w') as f:
            json.dump(self.history, f, indent=4)

def combined_loss_with_hla(hla):
    def loss(y_true, y_pred):
        current_weights = hla.get_weights()
        adjusted_weights = np.clip(current_weights, 1.0, 8.0)
        adjusted_weights[2] *= hla.tumor_boost
        ce_loss = tf.keras.losses.sparse_categorical_crossentropy(y_true, y_pred)
        weights = tf.cast(tf.gather(adjusted_weights, tf.cast(y_true, tf.int32)), tf.float32)
        weighted_ce = tf.reduce_mean(ce_loss * weights)
        focal = focal_loss(gamma=2., alpha=0.25)(y_true, y_pred)
        tversky = tversky_loss(y_true, y_pred, alpha=0.6, beta=0.4)
        return 0.25 * weighted_ce + 0.25 * focal + 0.5 * tversky
    return loss

def combined_loss_with_bla(bla):
    def loss(y_true, y_pred):
        current_weights = bla.get_weights()
        adjusted_weights = np.clip(current_weights, 1.0, 8.0)
        adjusted_weights[2] *= bla.tumor_boost
        ce_loss = tf.keras.losses.sparse_categorical_crossentropy(y_true, y_pred)
        weights = tf.cast(tf.gather(adjusted_weights, tf.cast(y_true, tf.int32)), tf.float32)
        weighted_ce = tf.reduce_mean(ce_loss * weights)
        focal = focal_loss(gamma=2., alpha=0.25)(y_true, y_pred)
        tversky = tversky_loss(y_true, y_pred, alpha=0.6, beta=0.4)
        return 0.25 * weighted_ce + 0.25 * focal + 0.5 * tversky
    return loss

def combined_loss_with_la(la):
    return combined_loss_with_bla(la)

def combined_loss_with_cla(cla):
    return combined_loss_with_bla(cla)
