import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
from google.colab import drive
drive.mount('/content/drive', force_remount=True)

# ============== Load Data ==============
data = {
    'Patient': ['volume-2', 'volume-7', 'volume-86', 'volume-87', 'volume-88', 
                'volume-90', 'volume-92', 'volume-93'],
    'Tumor_Slices': [0, 22, 5, 0, 13, 43, 5, 8],
    'Tumor_Pixels': [0, 10155, 785, 0, 26795, 61211, 172, 8373],
    'Liver_DSC': [0.992784, 0.985736, 0.991847, 0.938813, 0.946545, 
                  0.983919, 0.986952, 0.988679],
    'Tumor_DSC': [1.000000, 0.828575, 0.993421, 1.000000, 0.845409, 
                  0.916801, 0.600001, 0.834930],
    'Liver_ASSD': [0.007199, 0.014654, 0.008836, 0.073309, 0.017390, 
                   0.021233, 0.013808, 0.033714],
    'Tumor_ASSD': [0.000000, 0.090887, 0.006348, 0.000000, 0.032326, 
                   0.213328, 0.000000, 0.173430]
}

df = pd.DataFrame(data)

# ============== Journal-Quality Plot ==============
plt.style.use('default')
sns.set_style("whitegrid")
fig, axes = plt.subplots(2, 2, figsize=(15, 11))

# 1. Tumor DSC per Patient
sns.barplot(x='Patient', y='Tumor_DSC', data=df, ax=axes[0,0], palette='Blues_d')
axes[0,0].set_title('Tumor Dice Score per Patient', fontsize=13, fontweight='bold')
axes[0,0].set_ylabel('Tumor DSC')
axes[0,0].set_xticklabels(df['Patient'], rotation=45, ha='right')

# 2. Number of Tumor Slices + Pixels
ax2 = axes[0,1].twinx()
axes[0,1].bar(df['Patient'], df['Tumor_Slices'], color='#2E86AB', alpha=0.7, label='Number of Tumor Slices')
ax2.plot(df['Patient'], df['Tumor_Pixels']/1000, color='#A23B72', marker='o', linewidth=2.5, label='Tumor Pixels (×1000)')
axes[0,1].set_title('Tumor Burden per Patient', fontsize=13, fontweight='bold')
axes[0,1].set_ylabel('Number of Tumor Slices')
ax2.set_ylabel('Tumor Pixels (×1000)')
axes[0,1].legend(loc='upper left')
ax2.legend(loc='upper right')

# 3. Liver vs Tumor DSC
x = np.arange(len(df))
width = 0.35
axes[1,0].bar(x - width/2, df['Liver_DSC'], width, label='Liver DSC', color='#2CA02C')
axes[1,0].bar(x + width/2, df['Tumor_DSC'], width, label='Tumor DSC', color='#FF7F0E')
axes[1,0].set_title('Liver and Tumor DSC per Patient', fontsize=13, fontweight='bold')
axes[1,0].set_xticks(x)
axes[1,0].set_xticklabels(df['Patient'], rotation=45, ha='right')
axes[1,0].legend()

# 4. ASSD (Surface Distance)
axes[1,1].bar(x - width/2, df['Liver_ASSD'], width, label='Liver ASSD', color='#1F77B4')
axes[1,1].bar(x + width/2, df['Tumor_ASSD'], width, label='Tumor ASSD', color='#D62728')
axes[1,1].set_title('Average Symmetric Surface Distance (ASSD)', fontsize=13, fontweight='bold')
axes[1,1].set_ylabel('ASSD (mm)')
axes[1,1].set_xticks(x)
axes[1,1].set_xticklabels(df['Patient'], rotation=45, ha='right')
axes[1,1].legend()

plt.tight_layout()
plt.savefig("/content/drive/MyDrive/LITS/json/Figure_S2_PerPatient_Results.png", 
            dpi=300, bbox_inches='tight')
plt.show()
