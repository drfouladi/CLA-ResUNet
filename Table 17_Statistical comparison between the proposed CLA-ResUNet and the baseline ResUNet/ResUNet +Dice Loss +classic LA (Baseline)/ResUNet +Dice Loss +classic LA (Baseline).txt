# ====================== Metric Functions ======================
def dice_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(tf.argmax(y_pred, axis=-1) == class_id, tf.float32)
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    denominator = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f)
    return tf.where(denominator > 0, (2. * intersection + 1e-5) / (denominator + 1e-5), 0.0)

def iou_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(tf.argmax(y_pred, axis=-1) == class_id, tf.float32)
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f) - intersection
    return tf.where(union > 0, (intersection + 1e-5) / (union + 1e-5), 0.0)

def precision_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(tf.argmax(y_pred, axis=-1) == class_id, tf.float32)
    tp = tf.reduce_sum(y_true_f * y_pred_f)
    pred_pos = tf.reduce_sum(y_pred_f)
    return tf.where(pred_pos > 0, tp / (pred_pos + 1e-5), 0.0)

def recall_per_class(y_true, y_pred, class_id):
    y_true_f = tf.cast(y_true == class_id, tf.float32)
    y_pred_f = tf.cast(tf.argmax(y_pred, axis=-1) == class_id, tf.float32)
    tp = tf.reduce_sum(y_true_f * y_pred_f)
    actual_pos = tf.reduce_sum(y_true_f)
    return tf.where(actual_pos > 0, tp / (actual_pos + 1e-5), 0.0)

def dice_liver(y_true, y_pred): return dice_per_class(y_true, y_pred, 1)
def dice_tumor(y_true, y_pred): return dice_per_class(y_true, y_pred, 2)
def iou_liver(y_true, y_pred): return iou_per_class(y_true, y_pred, 1)
def iou_tumor(y_true, y_pred): return iou_per_class(y_true, y_pred, 2)
def precision_liver(y_true, y_pred): return precision_per_class(y_true, y_pred, 1)
def precision_tumor(y_true, y_pred): return precision_per_class(y_true, y_pred, 2)
def recall_liver(y_true, y_pred): return recall_per_class(y_true, y_pred, 1)
def recall_tumor(y_true, y_pred): return recall_per_class(y_true, y_pred, 2)
def dice_avg(y_true, y_pred):
    return (dice_liver(y_true, y_pred) + dice_tumor(y_true, y_pred)) / 2

# ====================== Loss Functions ======================
def dice_loss(y_true, y_pred):
    y_true_f = tf.one_hot(tf.cast(y_true, tf.int32), depth=3)
    y_pred_f = tf.nn.softmax(y_pred, axis=-1)
    intersection = tf.reduce_sum(y_true_f * y_pred_f, axis=[1, 2])
    union = tf.reduce_sum(y_true_f + y_pred_f, axis=[1, 2])
    dice = (2. * intersection + 1e-5) / (union + 1e-5)
    return tf.where(union > 0, 1 - tf.reduce_mean(dice), 0.0)

def focal_loss(gamma=2., alpha=0.25):
    def focal_loss_fixed(y_true, y_pred):
        y_true = tf.one_hot(tf.cast(y_true, tf.int32), depth=3)
        y_pred = tf.clip_by_value(y_pred, 1e-7, 1. - 1e-7)
        ce = -y_true * tf.math.log(y_pred)
        weight = y_true * tf.pow(1 - y_pred, gamma)
        fl = alpha * weight * ce
        return tf.reduce_mean(fl)
    return focal_loss_fixed

# ====================== Data ======================
def load_and_preprocess(vol_path, seg_path):
    try:
        vol_data = np.load(vol_path)
        seg_data = np.load(seg_path)
        vol_resized = tf.image.resize(vol_data[..., np.newaxis], (128, 128)).numpy()
        seg_resized = tf.image.resize(seg_data[..., np.newaxis], (128, 128), method='nearest').numpy()
        vol_normalized = (vol_resized - vol_resized.min()) / (vol_resized.max() - vol_resized.min() + 1e-6)
        seg_resized = np.round(seg_resized).astype(np.uint8)
        seg_resized = np.clip(seg_resized[..., 0], 0, 2)
        return vol_normalized, seg_resized
    except Exception as e:
        print(f"Error loading {vol_path}: {e}")
        return np.zeros((128, 128, 1)), np.zeros((128, 128))

def data_generator(pairs, batch_size=8):
    while True:
        np.random.shuffle(pairs)
        for i in range(0, len(pairs), batch_size):
            batch_pairs = pairs[i:i + batch_size]
            vol_batch = []
            seg_batch = []
            for vol_path, seg_path in batch_pairs:
                vol_norm, seg = load_and_preprocess(vol_path, seg_path)
                vol_batch.append(vol_norm)
                seg_batch.append(seg)
            if len(vol_batch) == 0:
                continue
            yield (np.array(vol_batch), np.array(seg_batch))

# ====================== ResUNet ======================
def residual_block(x, filters, kernel_size=3, strides=1):
    shortcut = x
    x = layers.Conv2D(filters, kernel_size, strides=strides, padding='same')(x)
    x = layers.BatchNormalization()(x)
    x = layers.Activation('relu')(x)
    x = layers.Conv2D(filters, kernel_size, padding='same')(x)
    x = layers.BatchNormalization()(x)
    if strides != 1 or shortcut.shape[-1] != filters:
        shortcut = layers.Conv2D(filters, 1, strides=strides, padding='same')(shortcut)
        shortcut = layers.BatchNormalization()(shortcut)
    x = layers.Add()([shortcut, x])
    x = layers.Activation('relu')(x)
    return x

def resunet(input_shape=(128, 128, 1), num_classes=3):
    inputs = layers.Input(input_shape)
    c1 = residual_block(inputs, 64)
    p1 = layers.MaxPooling2D((2, 2))(c1)
    c2 = residual_block(p1, 128)
    p2 = layers.MaxPooling2D((2, 2))(c2)
    c3 = residual_block(p2, 256)
    p3 = layers.MaxPooling2D((2, 2))(c3)
    c4 = residual_block(p3, 512)
    p4 = layers.MaxPooling2D((2, 2))(c4)
    bridge = residual_block(p4, 1024)
    u4 = layers.UpSampling2D((2, 2))(bridge)
    u4 = layers.Concatenate()([u4, c4])
    d4 = residual_block(u4, 512)
    u3 = layers.UpSampling2D((2, 2))(d4)
    u3 = layers.Concatenate()([u3, c3])
    d3 = residual_block(u3, 256)
    u2 = layers.UpSampling2D((2, 2))(d3)
    u2 = layers.Concatenate()([u2, c2])
    d2 = residual_block(u2, 128)
    u1 = layers.UpSampling2D((2, 2))(d2)
    u1 = layers.Concatenate()([u1, c1])
    d1 = residual_block(u1, 64)
    outputs = layers.Conv2D(num_classes, 1, activation='softmax')(d1)
    return Model(inputs, outputs)

# ====================== Post-processing ======================
def post_process_predictions(pred_probs, tumor_threshold=0.35, liver_min_size=30, tumor_min_size=3):
    pred_labels = np.argmax(pred_probs, axis=-1).astype(np.uint8)
    tumor_probs = pred_probs[..., 2]
    pred_labels[(pred_labels == 2) & (tumor_probs < tumor_threshold)] = 0
    for i in range(pred_labels.shape[0]):
        for mask, min_size, class_id in [(pred_labels[i] == 1, liver_min_size, 1), (pred_labels[i] == 2, tumor_min_size, 2)]:
            labeled, num = measure.label(mask.astype(np.uint8), return_num=True)
            for label in range(1, num + 1):
                if np.sum(labeled == label) < min_size:
                    pred_labels[i][labeled == label] = 0
    return pred_labels

# ====================== calculate_metrics ======================
def calculate_metrics(model, val_gen, num_samples):
    def compute_metrics(gt, pred):
        gt = gt.astype(np.uint8)
        pred = pred.astype(np.uint8)
        metrics = {}
        for cls, name in [(1, "Liver"), (2, "Tumor")]:
            gt_bin = (gt == cls).astype(np.float32)
            pred_bin = (pred == cls).astype(np.float32)
            tp = np.sum(gt_bin * pred_bin)
            fp = np.sum(pred_bin) - tp
            fn = np.sum(gt_bin) - tp

            dice = (2 * tp + 1e-5) / (np.sum(gt_bin) + np.sum(pred_bin) + 1e-5)
            iou = (tp + 1e-5) / (tp + fp + fn + 1e-5)
            precision = tp / (tp + fp + 1e-5)
            recall = tp / (tp + fn + 1e-5)
            voe = 1 - iou
            rvd = (np.sum(pred_bin) - np.sum(gt_bin)) / (np.sum(gt_bin) + 1e-5)

            gt_pts = np.argwhere(gt_bin > 0)
            pred_pts = np.argwhere(pred_bin > 0)
            if len(gt_pts) > 0 and len(pred_pts) > 0:
                d1 = cdist(gt_pts, pred_pts)
                assd = (np.mean(np.min(d1, axis=1)) + np.mean(np.min(d1, axis=0))) / 2
                msd = np.max(np.min(d1, axis=1))
            else:
                assd = 0.0
                msd = 0.0

            metrics[f"{name}_DSC"] = dice
            metrics[f"{name}_IoU"] = iou
            metrics[f"{name}_Precision"] = precision
            metrics[f"{name}_Recall"] = recall
            metrics[f"{name}_VOE"] = voe
            metrics[f"{name}_RVD"] = rvd
            metrics[f"{name}_ASSD"] = assd
            metrics[f"{name}_MSD"] = msd

        return metrics

    all_metrics = []
    for i in range(num_samples):
        try:
            images, true_labels = next(val_gen)
            pred_probs = model.predict(images, verbose=0)
            pred_labels = post_process_predictions(pred_probs)
            for j in range(len(true_labels)):
                m = compute_metrics(true_labels[j], pred_labels[j])
                all_metrics.append(m)
        except Exception as e:
            print(f"Error batch {i}: {e}")

    df = pd.DataFrame(all_metrics)

    print("\n===== Final Metrics (Baseline) =====")
    for col in df.columns:
        mean_val = df[col].mean()
        std_val = df[col].std()
        print(f"{col}: {mean_val:.4f} ± {std_val:.4f}")

    df.to_csv("/content/drive/MyDrive/LITS/baseline_per_slice_metrics.csv", index=False)
    print(f"Per-slice metrics saved ({len(df)})")
    return df

    except:
        continue

# ====================== Model ======================
model = resunet(input_shape=(128, 128, 1), num_classes=3)
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=4.2e-5),
    loss=dice_loss,
    metrics=['accuracy', dice_liver, dice_tumor, iou_liver, iou_tumor,
             precision_liver, precision_tumor, recall_liver, recall_tumor, dice_avg]

# ====================== Training ======================
history = model.fit(

    steps_per_epoch=len(train_pairs) // 8,
    validation_data=val_gen,
    validation_steps=len(val_pairs) // 8,
    epochs=100,
    callbacks=[
        callbacks.EarlyStopping(monitor='val_dice_tumor', patience=20, restore_best_weights=True, mode='max'),

    ]
