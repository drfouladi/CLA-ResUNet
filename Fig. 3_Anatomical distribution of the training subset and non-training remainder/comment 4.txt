# ============================================================
# FINAL METADATA-BASED REPRESENTATIVENESS ANALYSIS
# ============================================================
# IMPORTANT:
# - Does NOT modify training code
# - Does NOT modify model
# - Does NOT retrain anything
# - Uses ALL 19,163 metadata records
# - Anatomical and tumor-size analysis is based on metadata only
# - Training subset is reconstructed using the same sampling/split logic
# ============================================================

import os
import json
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from scipy.stats import chi2_contingency, ks_2samp, mannwhitneyu
from sklearn.model_selection import train_test_split


# ============================================================
# 1. PATHS
# ============================================================

BASE_PATH = "/content/drive/MyDrive/LITS"

METADATA_PATH = os.path.join(
    BASE_PATH,
    "json",
    "slice_metadata.json"
)

OUTPUT_PATH = os.path.join(
    BASE_PATH,
    "Final_Metadata_Representativeness"
)

os.makedirs(OUTPUT_PATH, exist_ok=True)

print("=" * 75)
print("FINAL METADATA-BASED REPRESENTATIVENESS ANALYSIS")
print("=" * 75)

print("Metadata:", METADATA_PATH)
print("Output:", OUTPUT_PATH)


# ============================================================
# 2. LOAD COMPLETE METADATA
# ============================================================

with open(METADATA_PATH, "r", encoding="utf-8") as f:
    metadata = json.load(f)

df = pd.DataFrame(metadata)

print("\nTotal metadata records:", len(df))
print("Unique volumes:", df["volume_id"].nunique())


# ============================================================
# 3. BASIC DATA VALIDATION
# ============================================================

required_columns = [
    "slice_id",
    "volume_id",
    "volume_name",
    "original_slice",
    "contains_liver",
    "contains_tumor",
    "liver_pixels",
    "tumor_pixels",
    "slice_filename",
    "slice_index_saved"
]

missing_columns = [
    c for c in required_columns
    if c not in df.columns
]

if missing_columns:
    raise ValueError(
        f"Missing required metadata columns: {missing_columns}"
    )

print("\nAll required metadata fields are present.")


# ============================================================
# 4. NORMALIZE BOOLEAN VARIABLES
# ============================================================

df["contains_tumor"] = (
    df["contains_tumor"]
    .astype(bool)
)

df["contains_liver"] = (
    df["contains_liver"]
    .astype(bool)
)


# ============================================================
# 5. WITHIN-VOLUME NORMALIZED ANATOMICAL POSITION
# ============================================================
# IMPORTANT:
# Anatomical position is calculated separately for each volume.
# This avoids bias caused by different numbers of slices per volume.

group_min = (
    df.groupby("volume_id")["original_slice"]
    .transform("min")
)

group_max = (
    df.groupby("volume_id")["original_slice"]
    .transform("max")
)

denominator = group_max - group_min

df["normalized_anatomical_position"] = np.where(
    denominator > 0,
    (df["original_slice"] - group_min) / denominator,
    0.5
)


# ============================================================
# 6. FIVE ANATOMICAL REGIONS
# ============================================================

bins = [-0.001, 0.20, 0.40, 0.60, 0.80, 1.001]

region_labels = [
    "Region 1 (0–20%)",
    "Region 2 (20–40%)",
    "Region 3 (40–60%)",
    "Region 4 (60–80%)",
    "Region 5 (80–100%)"
]

df["anatomical_region"] = pd.cut(
    df["normalized_anatomical_position"],
    bins=bins,
    labels=region_labels,
    include_lowest=True
)


# ============================================================
# 7. RECONSTRUCT THE ORIGINAL 500 + 500 SAMPLING
# ============================================================
# This follows the same logic as the working training code:
#
# tumor_pairs = tumor_pairs[:500]
# other_pairs = other_pairs[:500]
# pairs_selected = tumor_pairs + other_pairs
#
# Then:
#
# train_test_split(
#     pairs_selected,
#     test_size=0.2,
#     random_state=42
# )
#
# We reproduce this using slice_index_saved as the stable identifier.

tumor_df = df[
    df["contains_tumor"] == True
].copy()

non_tumor_df = df[
    df["contains_tumor"] == False
].copy()

print("\nTumor-containing metadata records:",
      len(tumor_df))

print("Non-tumor metadata records:",
      len(non_tumor_df))


# Sort to reproduce deterministic first-500 selection
tumor_df = tumor_df.sort_values(
    "slice_index_saved"
).reset_index(drop=True)

non_tumor_df = non_tumor_df.sort_values(
    "slice_index_saved"
).reset_index(drop=True)


tumor_selected = tumor_df.head(500).copy()

non_tumor_selected = non_tumor_df.head(500).copy()


selected = pd.concat(
    [
        tumor_selected,
        non_tumor_selected
    ],
    ignore_index=True
)


print("\nSelected tumor slices:",
      len(tumor_selected))

print("Selected non-tumor slices:",
      len(non_tumor_selected))

print("Total selected:",
      len(selected))


# ============================================================
# 8. ORIGINAL 80/20 TRAIN/VALIDATION SPLIT
# ============================================================

train_df, val_df = train_test_split(
    selected,
    test_size=0.2,
    random_state=42
)

train_df = train_df.copy()
val_df = val_df.copy()

train_ids = set(
    train_df["slice_index_saved"]
)

val_ids = set(
    val_df["slice_index_saved"]
)

print("\nTraining slices:", len(train_df))
print("Validation slices:", len(val_df))


# ============================================================
# 9. LABEL ALL 19,163 METADATA RECORDS
# ============================================================

df["group"] = "Remainder"

df.loc[
    df["slice_index_saved"].isin(train_ids),
    "group"
] = "Training"

df.loc[
    df["slice_index_saved"].isin(val_ids),
    "group"
] = "Validation"


print("\nGroup distribution:")
print(
    df["group"]
    .value_counts()
)


# ============================================================
# 10. TUMOR-SIZE CATEGORIES
# ============================================================
# Empirical tertiles based on ALL tumor-containing metadata records.
#
# These are NOT clinical thresholds.

tumor_all = df[
    df["contains_tumor"] == True
].copy()

q33 = tumor_all[
    "tumor_pixels"
].quantile(1/3)

q66 = tumor_all[
    "tumor_pixels"
].quantile(2/3)

print("\nTumor-size percentile thresholds:")
print("33rd percentile:", q33)
print("66th percentile:", q66)


def size_category(x):
    if x <= q33:
        return "Small"
    elif x <= q66:
        return "Medium"
    else:
        return "Large"


tumor_all["tumor_size_category"] = (
    tumor_all["tumor_pixels"]
    .apply(size_category)
)


# ============================================================
# 11. TUMOR SIZE DISTRIBUTION
# ============================================================

size_table = pd.crosstab(
    tumor_all["tumor_size_category"],
    tumor_all["group"]
)

size_table = size_table.reindex(
    ["Small", "Medium", "Large"],
    fill_value=0
)

# Only Training vs Remainder for inferential comparison
size_inferential = size_table[
    ["Training", "Remainder"]
]

size_percent = (
    size_inferential
    .div(
        size_inferential.sum(axis=0),
        axis=1
    )
    * 100
)

chi2_size, p_size, df_size, expected_size = (
    chi2_contingency(size_inferential)
)

cramers_v_size = np.sqrt(
    chi2_size /
    (
        size_inferential.values.sum()
        * min(
            size_inferential.shape[0] - 1,
            size_inferential.shape[1] - 1
        )
    )
)

print("\n" + "=" * 75)
print("TUMOR SIZE DISTRIBUTION")
print("=" * 75)

print(size_percent.round(2))

print(
    f"\nChi-square = {chi2_size:.4f}"
)

print(
    f"df = {df_size}"
)

print(
    f"p-value = {p_size:.8g}"
)

print(
    f"Cramer's V = {cramers_v_size:.4f}"
)


# ============================================================
# 12. TUMOR AREA — TRAINING VS REMAINDER
# ============================================================

tumor_train = tumor_all[
    tumor_all["group"] == "Training"
]["tumor_pixels"].astype(float)

tumor_remainder = tumor_all[
    tumor_all["group"] == "Remainder"
]["tumor_pixels"].astype(float)


ks_stat, ks_p = ks_2samp(
    tumor_train,
    tumor_remainder
)

mw_stat, mw_p = mannwhitneyu(
    tumor_train,
    tumor_remainder,
    alternative="two-sided"
)

pooled_sd = np.sqrt(
    (
        np.var(tumor_train, ddof=1)
        +
        np.var(tumor_remainder, ddof=1)
    ) / 2
)

smd = (
    (
        np.mean(tumor_train)
        -
        np.mean(tumor_remainder)
    )
    /
    pooled_sd
)

print("\nTumor area comparison:")
print(
    f"Training median = {np.median(tumor_train):.2f}"
)

print(
    f"Remainder median = {np.median(tumor_remainder):.2f}"
)

print(
    f"KS statistic = {ks_stat:.4f}"
)

print(
    f"KS p-value = {ks_p:.8g}"
)

print(
    f"Mann-Whitney p-value = {mw_p:.8g}"
)

print(
    f"SMD = {smd:.4f}"
)


# ============================================================
# 13. ANATOMICAL DISTRIBUTION
# ============================================================

anat_table = pd.crosstab(
    df["anatomical_region"],
    df["group"]
)

anat_table = anat_table.reindex(
    region_labels,
    fill_value=0
)

anat_inferential = anat_table[
    ["Training", "Remainder"]
]

anat_percent = (
    anat_inferential
    .div(
        anat_inferential.sum(axis=0),
        axis=1
    )
    * 100
)

chi2_anat, p_anat, df_anat, expected_anat = (
    chi2_contingency(anat_inferential)
)

cramers_v_anat = np.sqrt(
    chi2_anat /
    (
        anat_inferential.values.sum()
        *
        min(
            anat_inferential.shape[0] - 1,
            anat_inferential.shape[1] - 1
        )
    )
)

print("\n" + "=" * 75)
print("ANATOMICAL DISTRIBUTION")
print("=" * 75)

print(anat_percent.round(2))

print(
    f"\nChi-square = {chi2_anat:.4f}"
)

print(
    f"df = {df_anat}"
)

print(
    f"p-value = {p_anat:.8g}"
)

print(
    f"Cramer's V = {cramers_v_anat:.4f}"
)


# ============================================================
# 14. TUMOR PRESENCE BY ANATOMICAL REGION
# ============================================================

tumor_presence = pd.crosstab(
    df["anatomical_region"],
    df["contains_tumor"]
)

tumor_presence = tumor_presence.reindex(
    region_labels,
    fill_value=0
)

tumor_presence.columns = [
    "Non-tumor",
    "Tumor"
]

tumor_presence_percent = (
    tumor_presence
    .div(
        tumor_presence.sum(axis=1),
        axis=0
    )
    * 100
)

chi2_presence, p_presence, df_presence, expected_presence = (
    chi2_contingency(tumor_presence)
)

cramers_v_presence = np.sqrt(
    chi2_presence /
    (
        tumor_presence.values.sum()
        *
        min(
            tumor_presence.shape[0] - 1,
            tumor_presence.shape[1] - 1
        )
    )
)

print("\n" + "=" * 75)
print("TUMOR PRESENCE BY ANATOMICAL REGION")
print("=" * 75)

print(
    tumor_presence_percent.round(2)
)

print(
    f"\nChi-square = {chi2_presence:.4f}"
)

print(
    f"df = {df_presence}"
)

print(
    f"p-value = {p_presence:.8g}"
)

print(
    f"Cramer's V = {cramers_v_presence:.4f}"
)
